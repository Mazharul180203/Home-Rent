-- Trigger for units_count
CREATE TRIGGER tr_units_count
ON units
AFTER INSERT, DELETE
AS
BEGIN
    UPDATE p
    SET units_count = (SELECT COUNT(*) FROM units u WHERE u.building_id = p.id AND u.deleted_at IS NULL)
    FROM properties p
    WHERE p.id IN (SELECT building_id FROM inserted UNION SELECT building_id FROM deleted);
END;
GO

-- Audit trigger for access_logs
CREATE TRIGGER tr_access_log
ON users
AFTER INSERT, UPDATE, DELETE
AS
BEGIN
    SET NOCOUNT ON;
    INSERT INTO access_logs (user_id, action, entity_type, entity_id, details, ip_address)
    SELECT i.id, 'INSERT', 'users', i.id, '{"operation": "user_create"}', CURRENT_REQUEST_ID()
    FROM inserted i
    UNION ALL
    SELECT d.id, 'DELETE', 'users', d.id, '{"operation": "user_delete"}', CURRENT_REQUEST_ID()
    FROM deleted d;
END;
GO

-- View for financial summary
CREATE VIEW vw_financial_summary AS
SELECT 
    p.id AS property_id,
    p.name AS property_name,
    COUNT(u.id) AS total_units,
    SUM(CASE WHEN u.status = 'occupied' THEN 1 ELSE 0 END) AS occupied_units,
    SUM(pay.amount) AS total_revenue
FROM properties p
LEFT JOIN units u ON u.building_id = p.id AND u.deleted_at IS NULL
LEFT JOIN leases l ON l.unit_id = u.id AND l.deleted_at IS NULL
LEFT JOIN payments pay ON pay.lease_id = l.id AND pay.status = 'paid' AND pay.deleted_at IS NULL
GROUP BY p.id, p.name;
GO
